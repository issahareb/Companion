import React, { useState, useEffect, useRef } from 'react';
import { 
  Settings, Share2, Copy, Check, Search, Shield, 
  Backpack, Crosshair, Zap, Flame, Hexagon,
  LogOut, ExternalLink, Loader2, ChevronRight, AlertTriangle, User
} from 'lucide-react';

// --- ECHTE BUNGIE API KONFIGURATION ---
const BUNGIE_API_KEY = 'c95aec1b858f4122a4b1451c89b591ab';
const BUNGIE_CLIENT_ID = '51625';
const BUNGIE_AUTH_URL = `https://www.bungie.net/en/OAuth/Authorize?client_id=${BUNGIE_CLIENT_ID}&response_type=code`;

// --- MOCK DATEN (Bis das echte Inventar geladen wird) ---
const mockInventory = [
  { id: '1', name: 'Schicksalsbringer', type: 'Handfeuerwaffe', slot: 'kinetic', rarity: 'legendary', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/12e2e92c57ba8b9cc67776156e5df4e2.jpg' },
  { id: '2', name: 'Gjallarhorn', type: 'Raketenwerfer', slot: 'power', rarity: 'exotic', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/5cbcd5970c6dfcb208b534608c0f5f8b.jpg' },
  { id: '3', name: 'Zügelloser Flug', type: 'Maschinenpistole', slot: 'energy', rarity: 'legendary', power: 1805, icon: 'https://www.bungie.net/common/destiny2_content/icons/5a0c10816997cbda306db09b93b80e81.jpg' },
  { id: '4', name: 'Wurmgott-Kürass', type: 'Panzerhandschuhe', slot: 'arms', rarity: 'exotic', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/80e927c3f0b2f15591fc09520e791b72.jpg' },
];

const ITEM_SLOTS = [
  { id: 'kinetic', label: 'Kinetik', icon: Crosshair },
  { id: 'energy', label: 'Energie', icon: Zap },
  { id: 'power', label: 'Power', icon: Flame },
  { id: 'helmet', label: 'Helm', icon: Shield },
  { id: 'arms', label: 'Arme', icon: Shield },
  { id: 'chest', label: 'Brust', icon: Shield },
  { id: 'legs', label: 'Beine', icon: Shield },
  { id: 'class', label: 'Klasse', icon: Hexagon },
];

export default function App() {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [isAuthenticating, setIsAuthenticating] = useState(false);
  const [authError, setAuthError] = useState(null);
  const [accessToken, setAccessToken] = useState(null);
  
  const [destinyProfile, setDestinyProfile] = useState(null);
  const [isFetchingProfile, setIsFetchingProfile] = useState(false);
  
  const [activeTab, setActiveTab] = useState('inventory');
  const [searchQuery, setSearchQuery] = useState('');
  const [loadout, setLoadout] = useState({});
  
  // WICHTIG: Blocker für React Strict Mode (verhindert doppelten API Call)
  const hasAttemptedAuth = useRef(false);

  // --- OAUTH2 LOGIK ---
  useEffect(() => {
    const savedToken = localStorage.getItem('bungie_access_token');
    if (savedToken) {
      setAccessToken(savedToken);
      setIsAuthenticated(true);
      return;
    }

    const urlParams = new URLSearchParams(window.location.search);
    const authCode = urlParams.get('code');

    if (authCode && !hasAttemptedAuth.current) {
      hasAttemptedAuth.current = true;
      setIsAuthenticating(true);
      exchangeCodeForToken(authCode);
    }
  }, []);

  // --- ECHTEN DESTINY CHARAKTER LADEN ---
  useEffect(() => {
    if (isAuthenticated && accessToken && !destinyProfile) {
      fetchDestinyProfile();
    }
  }, [isAuthenticated, accessToken, destinyProfile]);

  const fetchDestinyProfile = async () => {
    setIsFetchingProfile(true);
    try {
      const memRes = await fetch('https://www.bungie.net/Platform/User/GetMembershipsForCurrentUser/', {
        headers: { 'X-API-Key': BUNGIE_API_KEY, 'Authorization': `Bearer ${accessToken}` }
      });
      const memData = await memRes.json();

      if (!memData.Response || !memData.Response.destinyMemberships || memData.Response.destinyMemberships.length === 0) {
        throw new Error("Kein Destiny 2 Account mit diesem Login verknüpft.");
      }

      const activeMembership = memData.Response.destinyMemberships.find(m => m.crossSaveOverride === m.membershipType) || memData.Response.destinyMemberships[0];

      const profileRes = await fetch(`https://www.bungie.net/Platform/Destiny2/${activeMembership.membershipType}/Profile/${activeMembership.membershipId}/?components=200`, {
        headers: { 'X-API-Key': BUNGIE_API_KEY, 'Authorization': `Bearer ${accessToken}` }
      });
      const profileData = await profileRes.json();

      const characters = profileData.Response.characters.data;
      const lastPlayedCharId = Object.keys(characters).sort((a,b) => new Date(characters[b].dateLastPlayed).getTime() - new Date(characters[a].dateLastPlayed).getTime())[0];
      const activeChar = characters[lastPlayedCharId];

      const classNames = { 0: 'TITAN', 1: 'JÄGER', 2: 'WARLOCK' };

      setDestinyProfile({
        name: `${activeMembership.bungieGlobalDisplayName}#${activeMembership.bungieGlobalDisplayNameCode.toString().padStart(4, '0')}`,
        emblem: `https://www.bungie.net${activeChar.emblemPath}`,
        emblemBackground: `https://www.bungie.net${activeChar.emblemBackgroundPath}`,
        light: activeChar.light,
        className: classNames[activeChar.classType] || 'HÜTER'
      });

    } catch (error) {
      console.error("API Error:", error);
      if (error.message.includes('Token') || error.message.includes('Unauthorized')) {
        handleLogout();
      } else {
        setAuthError(error.message);
      }
    } finally {
      setIsFetchingProfile(false);
    }
  };

  const exchangeCodeForToken = async (code) => {
    try {
      const response = await fetch('https://www.bungie.net/Platform/App/OAuth/Token/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/x-www-form-urlencoded',
          'X-API-Key': BUNGIE_API_KEY 
        },
        body: new URLSearchParams({
          grant_type: 'authorization_code',
          client_id: BUNGIE_CLIENT_ID,
          code: code,
        })
      });

      const data = await response.json();

      if (data.access_token) {
        localStorage.setItem('bungie_access_token', data.access_token);
        setAccessToken(data.access_token);
        setIsAuthenticated(true);
        window.history.replaceState({}, document.title, window.location.pathname);
      } else {
        throw new Error(data.error_description || "Token wurde von Bungie verweigert.");
      }
    } catch (error) {
      console.error("Auth Token Error:", error);
      setAuthError(error.message || "Netzwerkfehler beim Anmelden.");
    } finally {
      setIsAuthenticating(false);
    }
  };

  const handleLogin = () => {
    window.location.href = BUNGIE_AUTH_URL;
  };

  const handleLogout = () => {
    localStorage.removeItem('bungie_access_token');
    setAccessToken(null);
    setIsAuthenticated(false);
    setDestinyProfile(null);
    setLoadout({});
    hasAttemptedAuth.current = false;
  };

  const equipItem = (item) => {
    setLoadout(prev => {
      const newLoadout = { ...prev };
      if (item.rarity === 'exotic') {
        const isWeapon = ['kinetic', 'energy', 'power'].includes(item.slot);
        Object.values(newLoadout).forEach(equippedItem => {
          if (equippedItem.rarity === 'exotic') {
            const equippedIsWeapon = ['kinetic', 'energy', 'power'].includes(equippedItem.slot);
            if (isWeapon === equippedIsWeapon) {
              delete newLoadout[equippedItem.slot];
            }
          }
        });
      }
      newLoadout[item.slot] = item;
      return newLoadout;
    });
  };

  // --- LOGIN SCREEN ---
  if (!isAuthenticated) {
    return (
      <div className="min-h-screen bg-[#0A0A0B] text-gray-200 flex flex-col items-center justify-center p-6 selection:bg-white/20 relative overflow-hidden">
        <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-white/[0.02] rounded-full blur-[100px] pointer-events-none" />
        
        <div className="max-w-md w-full space-y-12 text-center z-10 animate-in fade-in zoom-in-95 duration-1000">
          <div className="space-y-4">
            <Hexagon className="w-16 h-16 mx-auto text-white/80" strokeWidth={1} />
            <h1 className="text-3xl font-light tracking-[0.2em] text-white">LOADOUT FORGE</h1>
            <p className="text-sm text-gray-500 font-light tracking-wide">VERBINDUNG ZU BUNGIE HERSTELLEN</p>
          </div>

          {authError && (
            <div className="bg-red-900/20 border border-red-500/50 text-red-400 p-4 flex items-start space-x-3 text-left">
              <AlertTriangle className="w-5 h-5 flex-shrink-0 mt-0.5" />
              <div className="text-sm">
                <strong>Login fehlgeschlagen:</strong><br/>
                {authError}
              </div>
            </div>
          )}

          <button 
            onClick={handleLogin}
            disabled={isAuthenticating}
            className="group relative inline-flex items-center justify-center px-8 py-4 w-full bg-white/5 border border-white/10 hover:border-white/30 hover:bg-white/10 transition-all duration-500 overflow-hidden"
          >
            <div className="absolute inset-0 w-0 bg-white/5 transition-all duration-[500ms] ease-out group-hover:w-full" />
            {isAuthenticating ? (
              <span className="flex items-center space-x-3 text-sm tracking-[0.1em]">
                <Loader2 className="w-4 h-4 animate-spin text-white/50" />
                <span className="text-white/70">LADE PROFILDATEN...</span>
              </span>
            ) : (
              <span className="flex items-center space-x-3 text-sm tracking-[0.1em] text-white/80 group-hover:text-white relative z-10">
                <span>MIT BUNGIE ANMELDEN</span>
                <ExternalLink className="w-4 h-4 opacity-50 group-hover:opacity-100 transition-opacity" />
              </span>
            )}
          </button>
        </div>
      </div>
    );
  }

  // --- MAIN DASHBOARD ---
  return (
    <div className="min-h-screen bg-[#0A0A0B] text-gray-300 font-sans selection:bg-white/20 flex flex-col md:flex-row overflow-hidden">
      
      {/* Sidebar */}
      <nav className="md:w-64 bg-[#0F0F11] border-b md:border-b-0 md:border-r border-white/5 p-6 flex flex-col z-20">
        <div className="flex items-center space-x-4 mb-12">
          <Hexagon className="w-8 h-8 text-white/80" strokeWidth={1.5} />
          <h1 className="text-sm font-light tracking-[0.2em] text-white/90 hidden md:block mt-1">FORGE</h1>
        </div>

        <div className="flex-grow flex flex-col space-y-2">
          <NavButton icon={Backpack} label="INVENTAR" active={activeTab === 'inventory'} onClick={() => setActiveTab('inventory')} />
          <NavButton icon={Shield} label="LOADOUT" active={activeTab === 'loadout'} onClick={() => setActiveTab('loadout')} />
        </div>

        <div className="mt-auto hidden md:block pt-8 border-t border-white/5">
          <button onClick={handleLogout} className="flex items-center space-x-3 text-xs tracking-wider text-gray-500 hover:text-white transition-colors w-full group">
            <LogOut className="w-4 h-4 group-hover:-translate-x-1 transition-transform" />
            <span>ABMELDEN</span>
          </button>
        </div>
      </nav>

      {/* Main Content */}
      <main className="flex-grow flex flex-col relative h-[calc(100vh-80px)] md:h-screen overflow-hidden">
        
        {/* Header - CHARAKTER ANZEIGE */}
        <header className="px-8 py-4 flex items-center justify-between z-10 border-b border-white/5 bg-[#0A0A0B]/80 backdrop-blur-md relative overflow-hidden">
          {destinyProfile && (
            <div className="absolute inset-0 opacity-20 bg-cover bg-center z-0" style={{ backgroundImage: `url(${destinyProfile.emblemBackground})` }} />
          )}

          <div className="relative w-full max-w-md group z-10">
            <Search className="absolute left-0 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-500 group-focus-within:text-white transition-colors" />
            <input 
              type="text" 
              placeholder="Mock-Inventar durchsuchen..." 
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="w-full bg-transparent border-b border-white/10 py-2 pl-8 pr-4 text-sm focus:outline-none focus:border-white/50 transition-all text-white placeholder-gray-500 tracking-wide"
            />
          </div>

          <div className="hidden md:flex items-center space-x-4 z-10">
            <div className="text-right">
              {isFetchingProfile ? (
                 <div className="text-xs text-gray-500 flex items-center space-x-2">
                   <Loader2 className="w-3 h-3 animate-spin" />
                   <span>Lade Charakter...</span>
                 </div>
              ) : destinyProfile ? (
                <>
                  <div className="text-xs text-gray-400 tracking-wider mb-0.5">{destinyProfile.name}</div>
                  <div className="text-sm tracking-widest text-white uppercase">{destinyProfile.className}</div>
                  <div className="text-xs text-[#E5C76B] font-mono mt-0.5">✧ {destinyProfile.light}</div>
                </>
              ) : (
                 <div className="text-sm tracking-widest text-gray-500">Profil wird geladen...</div>
              )}
            </div>
            
            <div className="w-12 h-12 border border-white/20 p-0.5 relative overflow-hidden group bg-[#141417] flex items-center justify-center shadow-lg">
              {destinyProfile?.emblem ? (
                <img src={destinyProfile.emblem} alt="Avatar" className="w-full h-full object-cover filter brightness-90 group-hover:brightness-110 transition-all duration-500" />
              ) : (
                <User className="w-5 h-5 text-gray-600" />
              )}
            </div>
          </div>
        </header>

        {/* Mock-Inventar & Loadout Builder */}
        <div className="flex-grow overflow-y-auto px-8 py-8 custom-scrollbar relative z-10">
          <div className="bg-green-900/10 border border-green-500/20 text-green-400 p-4 rounded-sm text-xs tracking-wide mb-8 animate-in fade-in">
            <strong>Erfolgreich Verbunden!</strong> Wenn du deinen Charakter oben rechts siehst, funktioniert die API perfekt. (Die Waffen hier sind aktuell noch Dummys).
          </div>

          {activeTab === 'inventory' && (
            <div className="space-y-12 animate-in fade-in slide-in-from-bottom-4 duration-700 ease-out">
              {['kinetic', 'energy', 'power', 'helmet', 'arms', 'chest', 'legs', 'class'].map(slot => {
                const itemsInSlot = mockInventory.filter(i => i.slot === slot);
                if (itemsInSlot.length === 0) return null;
                const slotInfo = ITEM_SLOTS.find(s => s.id === slot);

                return (
                  <div key={slot} className="space-y-4">
                    <div className="flex items-center space-x-3 border-b border-white/5 pb-2">
                      {slotInfo && <slotInfo.icon className="w-4 h-4 text-gray-500" strokeWidth={1.5} />}
                      <h3 className="text-xs font-light text-gray-400 tracking-[0.2em] uppercase">{slotInfo?.label}</h3>
                    </div>
                    <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-6 gap-4">
                      {itemsInSlot.map((item, index) => (
                        <div key={item.id} className="animate-in fade-in zoom-in-95 fill-mode-backwards" style={{ animationDelay: `${index * 50}ms`, animationDuration: '500ms' }}>
                          <ItemCard item={item} isEquipped={loadout[item.slot]?.id === item.id} onClick={() => equipItem(item)} />
                        </div>
                      ))}
                    </div>
                  </div>
                );
              })}
            </div>
          )}

          {activeTab === 'loadout' && (
             <div className="max-w-6xl mx-auto flex flex-col lg:flex-row gap-12 animate-in fade-in duration-700 ease-out">
             <div className="flex-grow space-y-6">
               <h2 className="text-xs font-light text-gray-400 tracking-[0.2em] uppercase border-b border-white/5 pb-2">Ausrüstung</h2>
               <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                 {ITEM_SLOTS.map((slotInfo, index) => {
                   const item = loadout[slotInfo.id];
                   return (
                     <div key={slotInfo.id} className="animate-in fade-in slide-in-from-left-4 fill-mode-backwards" style={{ animationDelay: `${index * 50}ms` }}>
                       <EquippedSlot slotInfo={slotInfo} item={item} onUnequip={() => {
                         const newLoadout = {...loadout};
                         delete newLoadout[slotInfo.id];
                         setLoadout(newLoadout);
                       }}/>
                     </div>
                   );
                 })}
               </div>
             </div>
           </div>
          )}
        </div>
      </main>

      {/* Mobile Menü (damit es auf dem Handy nicht kaputt geht) */}
      <div className="md:hidden bg-[#0F0F11] border-t border-white/5 flex items-center justify-around p-4 z-20">
        <MobileNavButton icon={Backpack} label="INVENTAR" active={activeTab === 'inventory'} onClick={() => setActiveTab('inventory')} />
        <MobileNavButton icon={Shield} label="LOADOUT" active={activeTab === 'loadout'} onClick={() => setActiveTab('loadout')} />
        <button onClick={handleLogout} className="flex flex-col items-center space-y-1.5 w-16 transition-colors duration-300 text-gray-600 hover:text-white">
          <LogOut className="w-5 h-5" strokeWidth={1.5} />
          <span className="text-[9px] tracking-wider">ABMELDEN</span>
        </button>
      </div>

      <style dangerouslySetInnerHTML={{__html: `
        .custom-scrollbar::-webkit-scrollbar { width: 4px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: rgba(255,255,255,0.1); }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover { background: rgba(255,255,255,0.2); }
      `}} />
    </div>
  );
}

// --- HILFSKOMPONENTEN ---
function NavButton({ icon: Icon, label, active, onClick }) {
  return (
    <button onClick={onClick} className={`flex items-center space-x-4 px-4 py-3 w-full text-left transition-all duration-300 group ${active ? 'text-white' : 'text-gray-500 hover:text-gray-300'}`}>
      <Icon className={`w-4 h-4 transition-transform duration-300 ${active ? 'scale-110 text-white' : 'group-hover:scale-110'}`} strokeWidth={1.5} />
      <span className="text-xs tracking-[0.15em]">{label}</span>
      {active && <ChevronRight className="w-3 h-3 ml-auto opacity-50" />}
    </button>
  );
}

function MobileNavButton({ icon: Icon, label, active, onClick }) {
  return (
    <button onClick={onClick} className={`flex flex-col items-center justify-center space-y-1.5 w-16 transition-colors duration-300 ${active ? 'text-white' : 'text-gray-600'}`}>
      <Icon className="w-5 h-5" strokeWidth={1.5} />
      <span className="text-[9px] tracking-wider">{label}</span>
    </button>
  );
}

function ItemCard({ item, isEquipped, onClick }) {
  const isExotic = item.rarity === 'exotic';
  const rarityColor = isExotic ? '#E5C76B' : '#7D5C96'; 
  return (
    <div onClick={onClick} className={`group cursor-pointer relative bg-[#141417] border border-white/5 overflow-hidden transition-all duration-500 ${isEquipped ? 'opacity-40 grayscale-[50%]' : 'hover:-translate-y-1 hover:border-white/20 hover:shadow-[0_8px_30px_rgba(0,0,0,0.5)]'}`}>
      <div className="h-0.5 w-full" style={{ backgroundColor: rarityColor }} />
      <div className="aspect-square relative overflow-hidden">
        <img src={item.icon} alt={item.name} className={`w-full h-full object-cover transition-transform duration-700 ease-out group-hover:scale-110 ${isEquipped ? '' : 'filter brightness-90 group-hover:brightness-110'}`} />
        <div className="absolute inset-0 bg-gradient-to-t from-black/80 via-black/20 to-transparent" />
      </div>
      <div className="p-3 relative z-10 bg-[#141417]">
        <h4 className="text-[11px] text-white font-medium truncate tracking-wide" style={{ color: isExotic ? rarityColor : 'white' }}>{item.name}</h4>
      </div>
    </div>
  );
}

function EquippedSlot({ slotInfo, item, onUnequip }) {
  const isExotic = item?.rarity === 'exotic';
  const rarityColor = isExotic ? '#E5C76B' : '#7D5C96';
  return (
    <div className="group relative bg-[#141417] border border-white/5 p-3 flex items-center space-x-4 transition-all duration-300 hover:border-white/20">
      <div className="w-14 h-14 bg-black border border-white/10 flex-shrink-0 relative overflow-hidden flex items-center justify-center">
        {item ? (
          <><img src={item.icon} alt={item.name} className="w-full h-full object-cover z-10" /><div className="absolute top-0 left-0 w-full h-0.5 z-20" style={{ backgroundColor: rarityColor }} /></>
        ) : <slotInfo.icon className="w-5 h-5 text-gray-700" strokeWidth={1} />}
      </div>
      <div className="flex-grow min-w-0">
        <div className="text-[9px] text-gray-500 uppercase tracking-widest mb-0.5">{slotInfo.label}</div>
        {item ? <div className="text-xs font-medium truncate" style={{ color: isExotic ? rarityColor : 'white' }}>{item.name}</div> : <div className="text-xs text-gray-600 font-light italic">Leer</div>}
      </div>
      {item && <button onClick={onUnequip} className="p-2 text-gray-600 hover:text-white opacity-0 group-hover:opacity-100 transition-all duration-300"><Crosshair className="w-4 h-4 rotate-45" /></button>}
    </div>
  );
}
