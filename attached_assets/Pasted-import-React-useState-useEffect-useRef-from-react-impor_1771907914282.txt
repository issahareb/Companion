import React, { useState, useEffect, useRef } from 'react';
import { 
  Settings, Share2, Copy, Check, Search, Shield, 
  Backpack, Crosshair, Zap, Flame, Hexagon,
  LogOut, ExternalLink, Loader2, ChevronRight, AlertTriangle, User,
  Sparkles, Activity
} from 'lucide-react';

// --- BUNGIE API KONFIGURATION ---
const BUNGIE_API_KEY = 'c95aec1b858f4122a4b1451c89b591ab';
const BUNGIE_CLIENT_ID = '51625';
const BUNGIE_AUTH_URL = `https://www.bungie.net/en/OAuth/Authorize?client_id=${BUNGIE_CLIENT_ID}&response_type=code`;

// --- MOCK DATEN ---
const mockInventory = [
  { id: '1', name: 'Schicksalsbringer', type: 'Handfeuerwaffe', slot: 'kinetic', rarity: 'legendary', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/12e2e92c57ba8b9cc67776156e5df4e2.jpg' },
  { id: '2', name: 'Gjallarhorn', type: 'Raketenwerfer', slot: 'power', rarity: 'exotic', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/5cbcd5970c6dfcb208b534608c0f5f8b.jpg' },
  { id: '3', name: 'Zügelloser Flug', type: 'Maschinenpistole', slot: 'energy', rarity: 'legendary', power: 1805, icon: 'https://www.bungie.net/common/destiny2_content/icons/5a0c10816997cbda306db09b93b80e81.jpg' },
  { id: '4', name: 'Wurmgott-Kürass', type: 'Panzerhandschuhe', slot: 'arms', rarity: 'exotic', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/80e927c3f0b2f15591fc09520e791b72.jpg' },
  { id: '5', name: 'Helm des 14. Heiligen', type: 'Helm', slot: 'helmet', rarity: 'exotic', power: 1800, icon: 'https://www.bungie.net/common/destiny2_content/icons/68c8f0003b1373510526eef40dff7ec2.jpg' },
  { id: '6', name: 'Gießerei-Brustschutz', type: 'Brustschutz', slot: 'chest', rarity: 'legendary', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/1217e997f7dc8826d57cd8db82046bc3.jpg' },
  { id: '7', name: 'Dünenwanderer', type: 'Beinschutz', slot: 'legs', rarity: 'exotic', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/e77c8e92fcaedfe9e8e50b904d1ef5e8.jpg' },
  { id: '8', name: 'Markeloses Zeichen', type: 'Klassengegenstand', slot: 'class', rarity: 'legendary', power: 1810, icon: 'https://www.bungie.net/common/destiny2_content/icons/769396e95b0ff72e5ddb018b3ffb7923.jpg' },
];

const ITEM_SLOTS = [
  { id: 'kinetic', label: 'Kinetik', icon: Crosshair },
  { id: 'energy', label: 'Energie', icon: Zap },
  { id: 'power', label: 'Power', icon: Flame },
  { id: 'helmet', label: 'Helm', icon: Shield },
  { id: 'arms', label: 'Arme', icon: Shield },
  { id: 'chest', label: 'Brust', icon: Shield },
  { id: 'legs', label: 'Beine', icon: Shield },
  { id: 'class', label: 'Klasse', icon: Hexagon },
];

export default function App() {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [isAuthenticating, setIsAuthenticating] = useState(false);
  const [authError, setAuthError] = useState(null);
  const [accessToken, setAccessToken] = useState(null);
  
  const [destinyProfile, setDestinyProfile] = useState(null);
  const [isFetchingProfile, setIsFetchingProfile] = useState(false);
  
  const [activeTab, setActiveTab] = useState('inventory'); // 'inventory', 'loadout', 'settings'
  const [searchQuery, setSearchQuery] = useState('');
  const [loadout, setLoadout] = useState({});
  const [shareLink, setShareLink] = useState('');
  const [copied, setCopied] = useState(false);
  
  const hasAttemptedAuth = useRef(false);

  // --- OAUTH2 LOGIK ---
  useEffect(() => {
    const savedToken = localStorage.getItem('bungie_access_token');
    if (savedToken) {
      setAccessToken(savedToken);
      setIsAuthenticated(true);
      return;
    }

    const urlParams = new URLSearchParams(window.location.search);
    const authCode = urlParams.get('code');

    if (authCode && !hasAttemptedAuth.current) {
      hasAttemptedAuth.current = true;
      setIsAuthenticating(true);
      exchangeCodeForToken(authCode);
    }
  }, []);

  // --- ECHTEN DESTINY CHARAKTER LADEN ---
  useEffect(() => {
    if (isAuthenticated && accessToken && !destinyProfile) {
      fetchDestinyProfile();
    }
  }, [isAuthenticated, accessToken, destinyProfile]);

  const fetchDestinyProfile = async () => {
    setIsFetchingProfile(true);
    try {
      const memRes = await fetch('https://www.bungie.net/Platform/User/GetMembershipsForCurrentUser/', {
        headers: { 'X-API-Key': BUNGIE_API_KEY, 'Authorization': `Bearer ${accessToken}` }
      });
      const memData = await memRes.json();

      if (!memData.Response || !memData.Response.destinyMemberships || memData.Response.destinyMemberships.length === 0) {
        throw new Error("Kein Destiny 2 Account mit diesem Login verknüpft.");
      }

      const activeMembership = memData.Response.destinyMemberships.find(m => m.crossSaveOverride === m.membershipType) || memData.Response.destinyMemberships[0];

      const profileRes = await fetch(`https://www.bungie.net/Platform/Destiny2/${activeMembership.membershipType}/Profile/${activeMembership.membershipId}/?components=200`, {
        headers: { 'X-API-Key': BUNGIE_API_KEY, 'Authorization': `Bearer ${accessToken}` }
      });
      const profileData = await profileRes.json();

      const characters = profileData.Response.characters.data;
      const lastPlayedCharId = Object.keys(characters).sort((a,b) => new Date(characters[b].dateLastPlayed).getTime() - new Date(characters[a].dateLastPlayed).getTime())[0];
      const activeChar = characters[lastPlayedCharId];

      const classNames = { 0: 'TITAN', 1: 'JÄGER', 2: 'WARLOCK' };

      setDestinyProfile({
        name: `${activeMembership.bungieGlobalDisplayName}#${activeMembership.bungieGlobalDisplayNameCode.toString().padStart(4, '0')}`,
        emblem: `https://www.bungie.net${activeChar.emblemPath}`,
        emblemBackground: `https://www.bungie.net${activeChar.emblemBackgroundPath}`,
        light: activeChar.light,
        className: classNames[activeChar.classType] || 'HÜTER'
      });

    } catch (error) {
      console.error("API Error:", error);
      if (error.message.includes('Token') || error.message.includes('Unauthorized')) {
        handleLogout();
      } else {
        setAuthError(error.message);
      }
    } finally {
      setIsFetchingProfile(false);
    }
  };

  const exchangeCodeForToken = async (code) => {
    try {
      const response = await fetch('https://www.bungie.net/Platform/App/OAuth/Token/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/x-www-form-urlencoded',
          'X-API-Key': BUNGIE_API_KEY 
        },
        body: new URLSearchParams({
          grant_type: 'authorization_code',
          client_id: BUNGIE_CLIENT_ID,
          code: code,
        })
      });

      const data = await response.json();

      if (data.access_token) {
        localStorage.setItem('bungie_access_token', data.access_token);
        setAccessToken(data.access_token);
        setIsAuthenticated(true);
        window.history.replaceState({}, document.title, window.location.pathname);
      } else {
        throw new Error(data.error_description || "Token wurde von Bungie verweigert.");
      }
    } catch (error) {
      console.error("Auth Token Error:", error);
      setAuthError(error.message || "Netzwerkfehler beim Anmelden.");
    } finally {
      setIsAuthenticating(false);
    }
  };

  const handleLogin = () => {
    window.location.href = BUNGIE_AUTH_URL;
  };

  const handleLogout = () => {
    localStorage.removeItem('bungie_access_token');
    setAccessToken(null);
    setIsAuthenticated(false);
    setDestinyProfile(null);
    setLoadout({});
    hasAttemptedAuth.current = false;
  };

  const equipItem = (item) => {
    setLoadout(prev => {
      const newLoadout = { ...prev };
      if (item.rarity === 'exotic') {
        const isWeapon = ['kinetic', 'energy', 'power'].includes(item.slot);
        Object.values(newLoadout).forEach(equippedItem => {
          if (equippedItem.rarity === 'exotic') {
            const equippedIsWeapon = ['kinetic', 'energy', 'power'].includes(equippedItem.slot);
            if (isWeapon === equippedIsWeapon) {
              delete newLoadout[equippedItem.slot];
            }
          }
        });
      }
      newLoadout[item.slot] = item;
      return newLoadout;
    });
  };

  const generateShareLink = () => {
    const itemIds = Object.values(loadout).map(item => item.id).join(',');
    setShareLink(`https://destiny-forge.app/l/${itemIds || 'empty'}`);
  };

  const copyToClipboard = () => {
    navigator.clipboard.writeText(shareLink).then(() => setCopied(true));
    setTimeout(() => setCopied(false), 2000);
  };

  const filteredInventory = mockInventory.filter(item => 
    item.name.toLowerCase().includes(searchQuery.toLowerCase()) || 
    item.type.toLowerCase().includes(searchQuery.toLowerCase())
  );

  // --- LOGIN SCREEN ---
  if (!isAuthenticated) {
    return (
      <div className="min-h-screen bg-[#050507] text-gray-200 flex flex-col items-center justify-center p-6 relative overflow-hidden">
        {/* Atmosphärischer Hintergrund */}
        <div className="absolute inset-0 bg-[radial-gradient(circle_at_50%_50%,_rgba(40,40,60,0.4)_0%,_rgba(5,5,7,1)_100%)] z-0" />
        <div className="absolute top-0 left-0 w-full h-1/2 bg-gradient-to-b from-indigo-900/10 to-transparent blur-3xl z-0" />
        
        <div className="max-w-md w-full space-y-12 text-center z-10 animate-in fade-in zoom-in-95 duration-1000">
          <div className="space-y-6 relative">
            <div className="w-24 h-24 mx-auto relative flex items-center justify-center">
              <div className="absolute inset-0 border border-white/20 rotate-45 animate-[spin_15s_linear_infinite]" />
              <div className="absolute inset-2 border border-white/10 -rotate-12 animate-[spin_20s_linear_infinite_reverse]" />
              <Hexagon className="w-12 h-12 text-white shadow-[0_0_30px_rgba(255,255,255,0.3)]" strokeWidth={1} />
            </div>
            <div>
              <h1 className="text-4xl font-light tracking-[0.3em] text-white uppercase drop-shadow-lg">Forge</h1>
              <p className="text-xs text-[#E5C76B] font-mono tracking-widest mt-2 uppercase">System_Link_Required</p>
            </div>
          </div>

          {authError && (
            <div className="bg-red-950/50 border-l-2 border-red-500 text-red-400 p-4 flex items-start space-x-3 text-left backdrop-blur-sm destiny-clip-sm">
              <AlertTriangle className="w-5 h-5 flex-shrink-0 mt-0.5 text-red-500" />
              <div className="text-sm tracking-wide">
                <strong className="uppercase text-red-300 text-xs tracking-widest block mb-1">Authorization Failed</strong>
                {authError}
              </div>
            </div>
          )}

          <div className="relative group">
            <div className="absolute -inset-1 bg-gradient-to-r from-indigo-500 to-[#E5C76B] rounded-lg blur opacity-20 group-hover:opacity-40 transition duration-1000 group-hover:duration-200" />
            <button 
              onClick={handleLogin}
              disabled={isAuthenticating}
              className="relative w-full bg-[#0A0A0B]/80 border border-white/20 hover:border-white p-5 flex items-center justify-center transition-all duration-300 backdrop-blur-xl destiny-clip group-hover:-translate-y-1"
            >
              {isAuthenticating ? (
                <span className="flex items-center space-x-3 text-sm tracking-[0.2em] text-[#E5C76B]">
                  <Loader2 className="w-5 h-5 animate-spin" />
                  <span>INITIALIZING...</span>
                </span>
              ) : (
                <span className="flex items-center space-x-3 text-sm tracking-[0.2em] text-white">
                  <span>BUNGIE.NET LOGIN</span>
                  <ExternalLink className="w-4 h-4 opacity-50 group-hover:opacity-100 group-hover:translate-x-1 transition-all" />
                </span>
              )}
            </button>
          </div>
        </div>
      </div>
    );
  }

  // --- MAIN DASHBOARD ---
  return (
    <div className="h-screen bg-[#050507] text-gray-300 font-sans selection:bg-indigo-500/30 flex flex-col md:flex-row overflow-hidden relative">
      
      {/* Globaler animierter Hintergrund */}
      <div className="fixed inset-0 pointer-events-none z-0">
        <div className="absolute inset-0 bg-[url('https://www.transparenttextures.com/patterns/stardust.png')] opacity-[0.03]" />
        <div className="absolute top-[-20%] left-[-10%] w-[50%] h-[50%] bg-indigo-900/20 blur-[120px] rounded-full" />
        <div className="absolute bottom-[-20%] right-[-10%] w-[50%] h-[50%] bg-[#522F65]/10 blur-[120px] rounded-full" />
      </div>

      {/* Desktop Sidebar */}
      <nav className="hidden md:flex w-72 bg-[#0A0A0C]/90 backdrop-blur-2xl border-r border-white/5 flex-col z-20 relative">
        <div className="p-8 flex items-center space-x-4">
          <div className="relative">
            <Hexagon className="w-10 h-10 text-white/90" strokeWidth={1} />
            <div className="absolute inset-0 bg-white/20 blur-md rounded-full" />
          </div>
          <div>
            <h1 className="text-sm font-light tracking-[0.3em] text-white uppercase leading-tight">Loadout</h1>
            <h1 className="text-sm font-bold tracking-[0.3em] text-white uppercase leading-tight">Forge</h1>
          </div>
        </div>

        <div className="flex-grow flex flex-col space-y-2 px-4 mt-4">
          <NavButton icon={Backpack} label="INVENTAR" active={activeTab === 'inventory'} onClick={() => setActiveTab('inventory')} />
          <NavButton icon={Shield} label="LOADOUT" active={activeTab === 'loadout'} onClick={() => setActiveTab('loadout')} />
          
          <div className="my-4 border-t border-white/5 mx-2" />
          
          <NavButton icon={Settings} label="SYSTEM" active={activeTab === 'settings'} onClick={() => setActiveTab('settings')} />
        </div>

        {/* Mini Profil Footer Sidebar */}
        <div className="p-6 border-t border-white/5 bg-gradient-to-t from-black/50 to-transparent">
          {destinyProfile && (
            <div className="flex items-center space-x-3">
              <img src={destinyProfile.emblem} alt="Emblem" className="w-10 h-10 border border-white/10" />
              <div className="overflow-hidden">
                <div className="text-[10px] text-[#E5C76B] font-mono tracking-widest truncate">{destinyProfile.name}</div>
                <div className="text-[10px] text-gray-500 uppercase tracking-widest">{destinyProfile.className}</div>
              </div>
            </div>
          )}
        </div>
      </nav>

      {/* Main Content Area (Scroll Fix) */}
      <main className="flex-1 flex flex-col h-full relative z-10 overflow-hidden">
        
        {/* Header (Sticky) */}
        <header className="flex-shrink-0 h-20 px-4 md:px-8 flex items-center justify-between border-b border-white/5 bg-[#0A0A0C]/80 backdrop-blur-xl relative overflow-hidden z-30">
          
          {/* Emblem Background Overlay */}
          {destinyProfile && (
             <div className="absolute inset-0 z-0 opacity-30">
               <div className="absolute inset-0 bg-gradient-to-r from-[#0A0A0C] via-transparent to-transparent z-10" />
               <img src={destinyProfile.emblemBackground} alt="bg" className="w-full h-full object-cover object-right filter blur-sm" />
             </div>
          )}

          {/* Search Bar */}
          <div className="relative w-full max-w-sm group z-10">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 group-focus-within:text-white transition-colors" />
            <input 
              type="text" 
              placeholder="Inventar durchsuchen..." 
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="w-full bg-black/40 border border-white/10 hover:border-white/30 focus:border-[#E5C76B] py-2.5 pl-10 pr-4 text-sm focus:outline-none transition-all text-white placeholder-gray-500 tracking-wide destiny-clip-sm"
            />
          </div>

          {/* Mobile Settings Icon & Profile */}
          <div className="flex items-center space-x-4 z-10">
            <button 
              className="md:hidden p-2 text-gray-400 hover:text-white transition-colors"
              onClick={() => setActiveTab('settings')}
            >
              <Settings className="w-5 h-5" />
            </button>
            
            <div className="hidden md:flex items-center space-x-4">
              <div className="text-right">
                {isFetchingProfile ? (
                   <div className="text-xs text-gray-400 flex items-center space-x-2">
                     <Loader2 className="w-3 h-3 animate-spin" />
                     <span className="tracking-widest uppercase">Connecting...</span>
                   </div>
                ) : destinyProfile ? (
                  <>
                    <div className="text-sm tracking-widest text-white uppercase font-bold drop-shadow-md">{destinyProfile.className}</div>
                    <div className="text-[11px] text-[#E5C76B] font-mono mt-0.5 flex items-center justify-end space-x-1">
                      <Sparkles className="w-3 h-3" />
                      <span>{destinyProfile.light}</span>
                    </div>
                  </>
                ) : null}
              </div>
            </div>
          </div>
        </header>

        {/* SCROLLABLE AREA - The crucial fix for scrolling */}
        <div className="flex-1 overflow-y-auto overflow-x-hidden custom-scrollbar pb-24 md:pb-8 relative">
          <div className="p-4 md:p-8 max-w-7xl mx-auto space-y-8">
            
            {/* Banner Alert */}
            <div className="bg-indigo-950/30 border-l-2 border-indigo-500 p-4 flex items-start space-x-3 backdrop-blur-md shadow-lg">
              <Activity className="w-5 h-5 text-indigo-400 flex-shrink-0" />
              <div>
                <h4 className="text-xs text-indigo-300 font-bold uppercase tracking-widest mb-1">Verbindung Aktiv</h4>
                <p className="text-sm text-indigo-100/70 leading-relaxed">
                  Dein Spielerprofil wurde erfolgreich über das Bungie.net OAuth2 Protokoll geladen. 
                  Das Inventar nutzt zurzeit simulierte Daten (Mock).
                </p>
              </div>
            </div>

            {/* TAB: INVENTORY */}
            {activeTab === 'inventory' && (
              <div className="space-y-12 animate-in fade-in slide-in-from-bottom-8 duration-700 ease-out">
                {['kinetic', 'energy', 'power', 'helmet', 'arms', 'chest', 'legs', 'class'].map(slot => {
                  const itemsInSlot = filteredInventory.filter(i => i.slot === slot);
                  if (itemsInSlot.length === 0) return null;
                  const slotInfo = ITEM_SLOTS.find(s => s.id === slot);

                  return (
                    <div key={slot} className="space-y-4">
                      <div className="flex items-center space-x-3 border-b border-white/10 pb-2 relative">
                        <div className="absolute bottom-0 left-0 w-1/4 h-[1px] bg-gradient-to-r from-white/50 to-transparent" />
                        {slotInfo && <slotInfo.icon className="w-4 h-4 text-gray-400" strokeWidth={1.5} />}
                        <h3 className="text-xs font-bold text-gray-300 tracking-[0.2em] uppercase">{slotInfo?.label}</h3>
                      </div>
                      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-4 md:gap-5">
                        {itemsInSlot.map((item, index) => (
                          <div key={item.id} className="animate-in fade-in zoom-in-95 fill-mode-backwards" style={{ animationDelay: `${index * 50}ms`, animationDuration: '500ms' }}>
                            <ItemCard item={item} isEquipped={loadout[item.slot]?.id === item.id} onClick={() => equipItem(item)} />
                          </div>
                        ))}
                      </div>
                    </div>
                  );
                })}
              </div>
            )}

            {/* TAB: LOADOUT */}
            {activeTab === 'loadout' && (
               <div className="flex flex-col lg:flex-row gap-8 animate-in fade-in duration-700 ease-out">
                 {/* Links: Slots */}
                 <div className="flex-grow space-y-6">
                   <div className="flex items-center justify-between border-b border-white/10 pb-2 relative">
                     <div className="absolute bottom-0 left-0 w-1/4 h-[1px] bg-gradient-to-r from-white/50 to-transparent" />
                     <h2 className="text-xs font-bold text-gray-300 tracking-[0.2em] uppercase">Ausrüstung</h2>
                   </div>
                   
                   <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                     {ITEM_SLOTS.map((slotInfo, index) => {
                       const item = loadout[slotInfo.id];
                       return (
                         <div key={slotInfo.id} className="animate-in fade-in slide-in-from-left-4 fill-mode-backwards" style={{ animationDelay: `${index * 50}ms` }}>
                           <EquippedSlot slotInfo={slotInfo} item={item} onUnequip={() => {
                             const newLoadout = {...loadout};
                             delete newLoadout[slotInfo.id];
                             setLoadout(newLoadout);
                           }}/>
                         </div>
                       );
                     })}
                   </div>
                 </div>

                 {/* Rechts: Teilen / Stats */}
                 <div className="lg:w-80 flex-shrink-0 space-y-8">
                   <div className="bg-[#0A0A0C]/60 backdrop-blur-md border border-white/10 p-6 destiny-clip-lg relative overflow-hidden group">
                     {/* Glow effect on hover */}
                     <div className="absolute inset-0 bg-gradient-to-br from-indigo-500/10 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-500" />
                     
                     <h2 className="text-xs font-bold text-gray-300 tracking-[0.2em] uppercase mb-4 flex items-center">
                       <Share2 className="w-4 h-4 mr-2 text-indigo-400" /> Network Share
                     </h2>
                     <p className="text-xs text-gray-400 leading-relaxed mb-6">
                       Erzeuge einen verschlüsselten Direktlink, um diese Zusammenstellung an andere Hüter zu übermitteln.
                     </p>
                     
                     <button 
                       onClick={generateShareLink} 
                       className="w-full bg-white/5 hover:bg-white/10 border border-white/20 hover:border-white text-white font-bold py-3 text-xs tracking-[0.2em] uppercase transition-all duration-300 destiny-clip-sm"
                     >
                       Link Generieren
                     </button>
                     
                     {shareLink && (
                       <div className="mt-4 animate-in fade-in slide-in-from-top-2 duration-300">
                         <div className="flex items-center bg-black/50 border border-white/10 p-1 destiny-clip-sm">
                           <input type="text" readOnly value={shareLink} className="bg-transparent text-[10px] text-gray-400 px-3 py-2 w-full focus:outline-none font-mono" />
                           <button onClick={copyToClipboard} className="p-2 hover:bg-white/10 transition-colors text-white">
                             {copied ? <Check className="w-4 h-4 text-[#E5C76B]" /> : <Copy className="w-4 h-4" />}
                           </button>
                         </div>
                       </div>
                     )}
                   </div>
                 </div>
               </div>
            )}

            {/* TAB: SETTINGS (NEU HINZUGEFÜGT!) */}
            {activeTab === 'settings' && (
              <div className="max-w-2xl mx-auto space-y-8 animate-in fade-in zoom-in-95 duration-500">
                <div className="flex items-center justify-between border-b border-white/10 pb-2 relative">
                  <div className="absolute bottom-0 left-0 w-1/4 h-[1px] bg-gradient-to-r from-white/50 to-transparent" />
                  <h2 className="text-xs font-bold text-gray-300 tracking-[0.2em] uppercase">Systemkonfiguration</h2>
                </div>

                <div className="bg-[#0A0A0C]/80 border border-white/10 p-6 destiny-clip-lg backdrop-blur-md space-y-6">
                  
                  {/* Account Info */}
                  <div>
                    <h3 className="text-[10px] text-gray-500 font-bold uppercase tracking-widest mb-4">Verbundener Account</h3>
                    {destinyProfile ? (
                      <div className="flex items-center space-x-4 bg-black/40 p-4 border border-white/5 destiny-clip-sm">

function ItemCard({ item, isEquipped, onClick }) {
  const isExotic = item.rarity === 'exotic';
  const rarityColor = isExotic ? '#CEAE33' : '#522F65'; 
  const hoverShadow = isExotic ? 'hover:shadow-[0_0_20px_rgba(206,174,51,0.4)]' : 'hover:shadow-[0_0_20px_rgba(82,47,101,0.4)]';

  return (
    <div 
      onClick={onClick} 
      className={`group cursor-pointer relative bg-[#0F0F11] border border-white/10 transition-all duration-500 destiny-clip-sm
        ${isEquipped ? 'opacity-30 grayscale-[70%] scale-95 border-white/30' : `hover:-translate-y-1 hover:border-white/40 ${hoverShadow}`}
      `}
    >
      {/* Rarity Top Bar - Thicker for Destiny feel */}
      <div className="h-1 w-full relative z-20" style={{ backgroundColor: rarityColor }} />
      
      <div className="aspect-square relative overflow-hidden bg-black">
        <img 
          src={item.icon} 
          alt={item.name} 
          className={`w-full h-full object-cover transition-transform duration-1000 ease-out group-hover:scale-110 ${isEquipped ? '' : 'filter brightness-90 group-hover:brightness-110'}`} 
        />
        {/* Dunkler Gradient für Lesbarkeit */}
        <div className="absolute inset-0 bg-gradient-to-t from-[#0F0F11] via-transparent to-transparent opacity-90" />
        
        {/* Power Level Overlay */}
        <div className="absolute bottom-1 right-1 font-mono text-[10px] text-[#E5C76B] font-bold bg-black/60 px-1 backdrop-blur-sm border border-white/10 rounded-sm z-20">
          {item.power}
        </div>
      </div>
      
      <div className="p-2.5 relative z-10 bg-[#0F0F11]">
        <h4 className="text-[10px] font-bold truncate tracking-wider" style={{ color: isExotic ? '#E5C76B' : 'white' }}>{item.name}</h4>
        <p className="text-[8px] text-gray-500 uppercase tracking-widest mt-0.5">{item.type}</p>
      </div>
    </div>
  );
}

function EquippedSlot({ slotInfo, item, onUnequip }) {
  const isExotic = item?.rarity === 'exotic';
  const rarityColor = isExotic ? '#CEAE33' : '#522F65';
  const bgGlow = isExotic ? 'group-hover:bg-[#CEAE33]/5' : 'group-hover:bg-[#522F65]/5';

  return (
    <div className={`group relative bg-[#0A0A0C]/80 backdrop-blur-sm border border-white/10 p-3 flex items-center space-x-4 transition-all duration-500 hover:border-white/30 destiny-clip-sm ${bgGlow}`}>
      <div className="w-16 h-16 bg-black border border-white/10 flex-shrink-0 relative flex items-center justify-center destiny-clip-sm overflow-hidden shadow-inner">
        {item ? (
          <>
            <img src={item.icon} alt={item.name} className="w-full h-full object-cover z-10 filter brightness-110" />
            <div className="absolute top-0 left-0 w-full h-1 z-20" style={{ backgroundColor: rarityColor }} />
          </>
        ) : (
          <slotInfo.icon className="w-6 h-6 text-white/20" strokeWidth={1} />
        )}
      </div>
      
      <div className="flex-grow min-w-0 py-1">
        <div className="text-[8px] text-gray-500 font-bold uppercase tracking-[0.2em] mb-1">{slotInfo.label}</div>
        {item ? (
          <>
            <div className="text-[11px] font-bold truncate tracking-wide drop-shadow-md" style={{ color: isExotic ? '#E5C76B' : 'white' }}>{item.name}</div>
            <div className="text-[10px] text-gray-400 font-mono mt-1 flex items-center">
               <Sparkles className="w-3 h-3 mr-1 text-[#E5C76B] opacity-70" /> {item.power}
            </div>
          </>
        ) : (
          <div className="text-[11px] text-gray-600 font-light tracking-wide mt-1">Empty Slot</div>
        )}
      </div>

      {item && (
        <button 
          onClick={onUnequip} 
          className="p-2.5 text-gray-500 hover:text-white opacity-0 group-hover:opacity-100 transition-all duration-300 bg-white/5 hover:bg-red-500/20 rounded-full mr-1"
        >
          <Crosshair className="w-4 h-4 rotate-45" />
        </button>
      )}
    </div>
  );
}
